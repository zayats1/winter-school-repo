LEARNING_RATE = 0.8         # High rate for deterministic FrozenLake

def watch_agent(qtable=None, delay=0.2):
    """
    Runs one episode visually.
    """
    # Note: render_mode="human" allows us to see the game pop up
    env = gym.make(ENV_NAME, render_mode="human", is_slippery=False) if ENV_NAME == "FrozenLake-v1" else gym.make(ENV_NAME, render_mode="human")

    state, info = env.reset()

    print("\n🎬 Simulation Started...")

    for step in range(MAX_STEPS):

        else:

def train_agent():
    """
    Trains the agent using Q-Learning.
    """
    # Note: We turn off slippery mode for easier debugging of logic
    if ENV_NAME == "FrozenLake-v1":
        env = gym.make(ENV_NAME, is_slippery=False, render_mode=None)
    else:
        env = gym.make(ENV_NAME, render_mode=None)

    state_size = env.observation_space.n
    action_size = env.action_space.n
    qtable = np.zeros((state_size, action_size))

    epsilon = EPSILON_START
    rewards_history = []

    print(f"🔄 Training {ENV_NAME} for {TRAIN_EPISODES} episodes...")

    for _ in tqdm(range(TRAIN_EPISODES)):
        state, info = env.reset()
        done = False
        episode_reward = 0

        for _ in range(MAX_STEPS):
            # 1. Choose Action (Epsilon-Greedy)
            if random.uniform(0, 1) < epsilon:
                action = env.action_space.sample() # Explore
            else:
                action = np.argmax(qtable[state, :]) # Exploit

            # 2. Execute Action
            new_state, reward, terminated, truncated, info = env.step(action)

            # 3. Update Q-Table (The Bellman Equation)
            current_q = qtable[state, action]
            max_future_q = np.max(qtable[new_state, :])

            new_q = current_q + LEARNING_RATE * (reward + DISCOUNT_RATE * max_future_q - current_q)
            qtable[state, action] = new_q

            state = new_state
            episode_reward += reward

            if terminated or truncated:
                break

        rewards_history.append(episode_reward)
        epsilon = max(EPSILON_MIN, epsilon - EPSILON_DECAY)

    env.close()
    save_plots(rewards_history, ENV_NAME)
    return qtable

🧠 Concept Review

1. The Q-Table

The qtable is essentially a cheat sheet the agent creates.

    Rows: Every possible state (where the elf is on the ice).

    Columns: Every possible action (Left, Down, Right, Up).

    Cells: The "Quality" (Q-value) of taking that action in that state.

Initially, the table is all zeros. As the agent falls in holes or finds the frisbee (goal), it updates the numbers using the Bellman Equation:
NewQ(s,a)=Q(s,a)+α[R+γmaxQ(s′,a′)−Q(s,a)]

Where:

    α (Alpha): Learning Rate (how much we accept the new information).

    γ (Gamma): Discount Factor (how much we care about future rewards vs immediate ones).

    R: Reward received.

2. Epsilon-Greedy Strategy

This handles the Exploration vs. Exploitation dilemma.

    High Epsilon: The agent is curious. It picks random actions to map out the world.

    Low Epsilon: The agent is greedy. It picks the best known action from the Q-Table.

    We decay epsilon over time so the agent starts curious but ends up confident.

💡 Why is_slippery=False?

In standard FrozenLake, the ice is slippery. If you choose "Right", there is a 33% chance you go Right, 33% you go Up, and 33% you go Down.

    If your agent isn't learning, always switch is_slippery=False first.

    If it solves the non-slippery version, your code is correct.

    Then, switch it back to True and tune your hyperparameters (usually requires more episodes and a lower learning rate) to solve the "real" environment.
